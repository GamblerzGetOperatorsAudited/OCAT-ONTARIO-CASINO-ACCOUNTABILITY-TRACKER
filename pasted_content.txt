function ocatApp() {
    return {
        view: 'gateway',
        signedIn: false,
        showReporter: false,
        showMatrix: false,
        reportStep: 1,
        agreements: [false, false, false, false],
        pillars: [
            { title: "No Malice & Anti-Defamation", text: "I agree to provide information without embellishment or willful misleading statements. OCAT is not associated with members who abuse the platform for defamation." },
            { title: "Honest & Truthful", text: "I certify that every statement is the absolute truth of my personal experiences. False reporting leads to immediate ban." },
            { title: "Proof of Breach", text: "I agree to provide evidence of implications regarding AGCO Registrar breaches when requested." },
            { title: "Respectful Community", text: "No harassing, bullying, or targeted arguing with other platform members." }
        ],
        casinoList: ["BetMGM Ontario", "FanDuel Ontario", "DraftKings", "Northstar Gaming", "Bet365", "888 Casino", "PointsBet", "LeoVegas"],
        standards: [
            { id: "Standard 4.10", title: "Game Malfunctions", desc: "Operators must have clear policies for handling technical glitches and ensuring fair outcomes." },
            { id: "Standard 5.01", title: "Asset Protection", desc: "Ensures that player funds and promised prizes (like extra spins) are accurately handled and protected." },
            { id: "Standard 2.01", title: "Responsible Gambling", desc: "Mandates that players are provided with tools to limit play and access support." }
        ],
        reports: [],
        newReport: { casino: '', desc: '', date: '', category: 'Integrity', answers: {} },
        questionGroups: [
            { title: "Step 2: Technical/Legal Audit", questions: [
                { id: 'q1', text: "Did the malfunction occur at a point of potential payout?" },
                { id: 'q2', text: "Was a Round ID or Game ID visible/provided?" },
                { id: 'q3', text: "Did you formally request Server-Side Game Logs?" },
                { id: 'q4', text: "Was that request for logs denied by the casino?" },
                { id: 'q5', text: "Did the T&Cs provided match the game's actual behavior?" }
            ]},
            { title: "Step 3: Process & Paper Trail", questions: [
                { id: 'q6', text: "Did a human (not a bot) respond within 24 hours?" },
                { id: 'q7', text: "Were you provided an Internal Dispute ID number?" },
                { id: 'q8', text: "Did the agent provide their 'Escalation Manager' info?" },
                { id: 'q9', text: "Was the interaction transcript shared with you?" },
                { id: 'q10', text: "Did they inform you of your right to contact iGO/AGCO?" }
            ]},
            { title: "Step 4: Retaliation & Ethics", questions: [
                { id: 'q11', text: "Was your account locked after you questioned them?" },
                { id: 'q12', text: "Are withdrawal delays happening alongside this dispute?" },
                { id: 'q13', text: "Did they offer a 'Goodwill Bonus' to drop the claim?" },
                { id: 'q14', text: "Did they ask you to sign a non-disclosure/waiver?" },
                { id: 'q15', text: "Did they blame your local internet/hardware?" }
            ]},
            { title: "Step 5: Impact & Health", questions: [
                { id: 'q16', text: "Is this incident causing financial or emotional distress?" },
                { id: 'q17', text: "Did you mention distress to the operator?" },
                { id: 'q18', text: "Did they offer RG tools after you showed distress?" },
                { id: 'q19', text: "Is this a recurring issue with this license holder?" },
                { id: 'q20', text: "Do you believe this is a systemic technical flaw?" }
            ]}
        ],
        init() {
            const saved = localStorage.getItem('ocat_reports');
            if (saved) {
                this.reports = JSON.parse(saved);
            } else {
                this.reports = [
                    { id: 'OCAT-001', casino: 'BetMGM Ontario', desc: 'Extra spins not finishing. Promised 10, only 6 delivered. Support denied server logs.', date: '2023-10-22', category: 'Standard 5.01' },
                    { id: 'OCAT-002', casino: 'Northstar Gaming', desc: 'Account lock immediately following 5k withdrawal request. Alleged verification loop.', date: '2023-10-24', category: 'Standard 4.1' }
                ];
            }
        },
        getStats() {
            let counts = {};
            this.reports.forEach(r => {
                counts[r.casino] = (counts[r.casino] || 0) + 1;
            });
            return Object.keys(counts).map(name => ({
                name,
                count: counts[name],
                percent: (counts[name] / this.reports.length) * 100
            })).sort((a,b) => b.count - a.count);
        },
        openReporter() {
            this.newReport = { casino: '', desc: '', date: '', category: 'Standard 5.01', answers: {} };
            this.reportStep = 1;
            this.showReporter = true;
        },
        openMatrix() {
            this.showMatrix = true;
        },
        nextStep() {
            if (this.reportStep < 5) {
                this.reportStep++;
            } else {
                this.submitReport();
            }
        },
        submitReport() {
            const id = 'OCAT-' + Math.floor(Math.random() * 1000).toString().padStart(3, '0');
            this.reports.unshift({...this.newReport, id: id});
            localStorage.setItem('ocat_reports', JSON.stringify(this.reports));
            this.showReporter = false;
            alert("ALLEGATION BROADCASTED: Your report is now part of the statistical database for public scrutiny.");
        }
    }
}
